function getCategories() {
  const cats = [...new Set(recipes.map(r => r.category).filter(Boolean))].sort();
  return ['All', ...cats];
}

function renderCategories() {
  const cats = getCategories();
  catFilter.innerHTML = cats.map(c => `
    <button class="cat-pill ${c === activeCategory ? 'active' : ''}" data-cat="${c}">${c}</button>
  `).join('');
  catFilter.querySelectorAll('.cat-pill').forEach(btn => {
    btn.addEventListener('click', () => {
      activeCategory = btn.dataset.cat;
      renderAll();
    });
  });
}

function getFiltered() {
  const q = searchInput.value.trim().toLowerCase();
  let list = recipes;

  if (activeCategory !== 'All') list = list.filter(r => r.category === activeCategory);
  if (q) list = list.filter(r =>
    r.name.toLowerCase().includes(q) ||
    r.category?.toLowerCase().includes(q) ||
    r.tags?.some(t => t.toLowerCase().includes(q)) ||
    r.ingredients?.toLowerCase().includes(q)
  );

  if (currentSort === 'newest') list = [...list].sort((a, b) => b.createdAt - a.createdAt);
  else if (currentSort === 'oldest') list = [...list].sort((a, b) => a.createdAt - b.createdAt);
  else if (currentSort === 'alpha') list = [...list].sort((a, b) => a.name.localeCompare(b.name));

  return list;
}

  tagsWrap.querySelectorAll('.tag-chip').forEach(c => c.remove());
  tags.forEach(t => {
    const chip = document.createElement('span');
    chip.className = 'tag-chip';
    chip.innerHTML = `${esc(t)}<button type="button" aria-label="Remove ${t}">✕</button>`;
    chip.querySelector('button').addEventListener('click', () => { tags = tags.filter(x => x !== t); renderTagChips(); });
    tagsWrap.insertBefore(chip, tagsInput);
  });

  if ((e.key === 'Enter' || e.key === ',') && tagsInput.value.trim()) {
    e.preventDefault();
    const v = tagsInput.value.trim().replace(/,$/, '');
    if (v && !tags.includes(v)) { tags.push(v); renderTagChips(); }
    tagsInput.value = '';
  }
  if (e.key === 'Backspace' && !tagsInput.value && tags.length) {
    tags.pop(); renderTagChips();
  }
});

function saveRecipe() {
  const name = document.getElementById('f_name').value.trim();
  const category = document.getElementById('f_category').value;
  const ingredients = document.getElementById('f_ingredients').value.trim();
  const steps = document.getElementById('f_steps').value.trim();

  if (!name) { showToast('Please enter a recipe name.'); return; }
  if (!category) { showToast('Please select a category.'); return; }
  if (!ingredients) { showToast('Please add some ingredients.'); return; }
  if (!steps) { showToast('Please add cooking steps.'); return; }

  // finalize any pending tag input
  const tagVal = tagsInput.value.trim();
  if (tagVal && !tags.includes(tagVal)) { tags.push(tagVal); tagsInput.value = ''; }

  const recipe = {
    id: editingId || uid(),
    name,
    category,
    emoji: selectedEmoji,
    prep: document.getElementById('f_prep').value.trim(),
    cook: document.getElementById('f_cook').value.trim(),
    servings: document.getElementById('f_servings').value.trim(),
    difficulty: document.getElementById('f_difficulty').value,
    ingredients,
    steps,
    notes: document.getElementById('f_notes').value.trim(),
    tags: [...tags],
    createdAt: editingId ? recipes.find(r => r.id === editingId)?.createdAt || Date.now() : Date.now(),
    updatedAt: Date.now()
  };

  if (editingId) {
    recipes = recipes.map(r => r.id === editingId ? recipe : r);
    showToast('Recipe updated! ✅');
  } else {
    recipes.unshift(recipe);
    showToast('Recipe saved! 🎉');
  }

  save();

  renderAll();
}

  showToast('Recipe deleted.');
  renderAll();
}

const IMPORT_SYSTEM = `You are a recipe extraction assistant. The user will provide either a URL to a social media video or recipe page, or raw text from a recipe. Extract the complete recipe and respond with ONLY a valid JSON object — no markdown, no explanation.

The JSON must have these fields (use null for missing info):
{
  "name": "Recipe name",
  "category": "One of: Breakfast, Lunch, Dinner, Snack, Dessert, Drinks, Soup, Salad, Bread & Baking, Vegetarian, Vegan",
  "emoji": "A single relevant food emoji",
  "prep": "e.g. 15 min or null",
  "cook": "e.g. 30 min or null",
  "servings": "e.g. 4 or null",
  "difficulty": "Easy, Medium, Hard, or null",
  "ingredients": "Each ingredient on its own line, plain text",
  "steps": "Each step on its own line, plain text, no numbering",
  "notes": "Any tips, substitutions, or storage notes, or null",
  "tags": ["tag1", "tag2"]
}

For social media URLs, infer the recipe from the URL context, hashtags, and any description you can determine. Be thorough — fill in as many fields as possible. Always produce a complete, usable recipe.`;

async function runImport() {
  importedRecipeData = null;
  importError.classList.remove('active');
  importPreview.classList.remove('active');
  importRunBtn.disabled = true;

  const isUrl = importMode === 'url';
  const input = isUrl ? importUrlInput.value.trim() : importTextInput.value.trim();

  if (!input) {
    showImportError(isUrl ? 'Please enter a URL.' : 'Please paste some recipe text.');
    return;
  }

  if (isUrl && !/^https?:\/\//i.test(input)) {
    showImportError('Please enter a valid URL starting with https://');
    return;
  }

  importLoading.classList.add('active');
  document.getElementById('importActions').style.display = 'none';

  const loadingMessages = isUrl
    ? ['Fetching recipe from URL…', 'Reading the page…', 'Extracting ingredients and steps…']
    : ['Reading your recipe text…', 'Structuring ingredients and steps…'];

  let msgIdx = 0;
  loadingMsg.textContent = loadingMessages[0];
  const msgInterval = setInterval(() => {
    msgIdx = (msgIdx + 1) % loadingMessages.length;
    loadingMsg.textContent = loadingMessages[msgIdx];
  }, 2200);

  const userPrompt = isUrl
    ? `Please extract the complete recipe from this URL: ${input}\n\nIf it's a social media video, infer the recipe from all available context (title, description, comments, hashtags). Return full recipe JSON.`
    : `Extract the complete recipe from this text and return it as JSON:\n\n${input}`;

  const tools = isUrl ? [{
    type: 'web_search_20250305',
    name: 'web_search'
  }] : undefined;

  try {
    const body = {
      model: 'claude-sonnet-4-20250514',
      max_tokens: 1000,
      system: IMPORT_SYSTEM,
      messages: [{ role: 'user', content: userPrompt }]
    };
    if (tools) body.tools = tools;

    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body)
    });

    clearInterval(msgInterval);

    if (!res.ok) {
      const err = await res.json().catch(() => ({}));
      throw new Error(err.error?.message || `API error ${res.status}`);
    }

    const data = await res.json();

    // Gather all text blocks (including after tool use)
    const textBlocks = (data.content || []).filter(b => b.type === 'text').map(b => b.text);
    const rawText = textBlocks.join('\n');

    // Extract JSON from the response
    const jsonMatch = rawText.match(/\{[\s\S]*\}/);
    if (!jsonMatch) throw new Error('Could not find recipe data in the response. Try again or paste the recipe text directly.');

    let recipe;
    try {
      recipe = JSON.parse(jsonMatch[0]);
    } catch {
      throw new Error('Recipe data was malformed. Please try again or paste the text directly.');
    }

    if (!recipe.name || !recipe.ingredients || !recipe.steps) {
      throw new Error('Could not extract a complete recipe. Try pasting the recipe text directly instead.');
    }

    importedRecipeData = recipe;
    importLoading.classList.remove('active');

    // Show preview
    document.getElementById('previewEmoji').textContent = recipe.emoji || '🍴';
    document.getElementById('previewName').textContent = recipe.name;
    document.getElementById('previewMeta').textContent = [
      recipe.category,
      recipe.prep && `Prep ${recipe.prep}`,
      recipe.cook && `Cook ${recipe.cook}`,
      recipe.servings && `Serves ${recipe.servings}`
    ].filter(Boolean).join(' · ');
    importPreview.classList.add('active');

    document.getElementById('importActions').style.display = '';
    importRunBtn.textContent = '→ Fill Recipe Form';
    importRunBtn.disabled = false;

  } catch (err) {
    clearInterval(msgInterval);
    showImportError('⚠️ ' + (err.message || 'Something went wrong. Please try again.'));
    document.getElementById('importActions').style.display = '';
  }
}

importRunBtn.addEventListener('click', () => {
  if (importedRecipeData) {
    // Fill the form with imported data
    closeDrawer(importDrawer);
    setTimeout(() => {
      clearForm();
      const r = importedRecipeData;
      document.getElementById('f_name').value = r.name || '';
      // Match category to select options
      const catSel = document.getElementById('f_category');
      const catOpts = [...catSel.options].map(o => o.value);
      catSel.value = catOpts.includes(r.category) ? r.category : '';
      document.getElementById('f_prep').value = r.prep || '';
      document.getElementById('f_cook').value = r.cook || '';
      document.getElementById('f_servings').value = r.servings || '';
      const diffSel = document.getElementById('f_difficulty');
      const diffOpts = [...diffSel.options].map(o => o.value);
      diffSel.value = diffOpts.includes(r.difficulty) ? r.difficulty : '';
      document.getElementById('f_ingredients').value = r.ingredients || '';
      document.getElementById('f_steps').value = r.steps || '';
      document.getElementById('f_notes').value = r.notes || '';
      // Set emoji
      const matchedEmoji = EMOJIS.includes(r.emoji) ? r.emoji : EMOJIS[0];
      selectedEmoji = matchedEmoji;

      renderTagChips();
      openDrawer(formDrawer);
      showToast('Recipe imported! Review and save. 🎉');
    }, 250);
  } else {
    runImport();
  }
});

if ('serviceWorker' in navigator) {
  window.addEventListener('load', () => {
    const swCode = `
const CACHE = 'recipe-book-v1';
const ASSETS = ['/'];

self.addEventListener('install', e => {
  e.waitUntil(caches.open(CACHE).then(c => c.addAll(ASSETS)).then(() => self.skipWaiting()));
});

self.addEventListener('activate', e => {
  e.waitUntil(
    caches.keys().then(keys =>
      Promise.all(keys.filter(k => k !== CACHE).map(k => caches.delete(k)))
    ).then(() => self.clients.claim())
  );
});

self.addEventListener('fetch', e => {
  if (e.request.method !== 'GET') return;
  e.respondWith(
    caches.match(e.request).then(cached => cached || fetch(e.request).then(res => {
      if (!res || res.status !== 200 || res.type === 'opaque') return res;
      const clone = res.clone();
      caches.open(CACHE).then(c => c.put(e.request, clone));
      return res;
    }))
  );
});
    `;
    const blob = new Blob([swCode], { type: 'application/javascript' });
    const swUrl = URL.createObjectURL(blob);
    navigator.serviceWorker.register(swUrl).catch(() => {
      // SW not critical — app works fine without it
    });
  });
}
